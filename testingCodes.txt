class Student:
    def average(self):
        if not self.marks:
            return 0
        return sum(self.marks) / len(self.marks)
students = [ Student("A", [82, 76, 91]),
    Student("B", [68, 74, 79]),
    Student("C", [95, 88, 92])]
for student in students:
    print(student.name, student.average())

def find_max(arr):
    # Handle empty list edge case
    if not arr:
        return None

    # Start by assuming the first element is the largest
    max_val = arr[0]

    # Check every other number in the array
    for num in arr[1:]:
        if num > max_val:
            max_val = num

    return max_val


# Quick test
numbers = [14, 3, 27, 8, 42, 19]
largest = find_max(numbers)
print("The maximum element is:", largest)

def find_pair_with_sum(nums, target):
    # Keep track of numbers we've seen and their index
    seen = {}

    for i, current_num in enumerate(nums):
        needed = target - current_num

        if needed in seen:
            return [seen[needed], i]

        seen[current_num] = i

    return []
